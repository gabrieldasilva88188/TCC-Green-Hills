===================
Nova Base de Dados - Base de Dados Que o Projeto se encontra atualmente
===================

-- Tabela de usuários
CREATE TABLE usuarios(
    id INT PRIMARY KEY AUTO_INCREMENT, 
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(11) NOT NULL,
    empresa VARCHAR(100) NOT NULL,
    user VARCHAR(15) NOT NULL, -- Tipo de usuário (adminMaster, admin, comum)
    email VARCHAR(100) NOT NULL,
    senha VARCHAR(100) NOT NULL
);

-- Tabela de estufas
CREATE TABLE estufas (
    id CHAR(8) PRIMARY KEY,
    nome_estufa VARCHAR(100) -- Nome opcional para identificação da estufa
);

-- Tabela para associar usuários e estufas (relacionamento muitos-para-muitos)
CREATE TABLE usuarios_estufas (
    usuario_id INT,
    estufa_id CHAR(8),
    PRIMARY KEY (usuario_id, estufa_id),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (estufa_id) REFERENCES estufas(id)
);

-- Tabela de sensores
CREATE TABLE tbsensores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    sensor1 VARCHAR(15),
    sensor2 VARCHAR(15),
    sensor3 VARCHAR(15),
    datahora DATETIME DEFAULT CURRENT_TIMESTAMP,
    estufa_id CHAR(8),
    FOREIGN KEY (estufa_id) REFERENCES estufas(id)
);
